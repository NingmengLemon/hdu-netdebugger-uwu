# 注意：里面的版本号必须与 pyproject.toml 的 version 保持一致 ——

    prodvers=(0, 1, 0, 0),

    mask=0x3f,
    # 文件属性的位掩码

    OS=0x40004,
    # 文件类型：0x1 = 应用程序
    fileType=0x1,
    # 该 fileType 下的用途：0x0 = 未定义
    subtype=0x0,
    # 创建时间戳（不填）
    date=(0, 0)
    ),
  kids=[
